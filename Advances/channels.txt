package main

import (
	"fmt"
	"time"
)

func main() {

	// variable := make(chan type) '<-' operator
	greeting := make(chan string)
	greetString := "Hello"

	go func() {
		greeting <- greetString // blocking because it's continuosly trying to receive values, it is ready to receive continuous flow of data
		greeting <- "World"
		for _, e := range "abcde" {
			greeting <- "Alphabet: " + string(e)
		}

	}()

	receiver := <-greeting
	fmt.Println(receiver)

	go func() {
		receiver := <-greeting
		fmt.Println(receiver)
		receiver = <-greeting
		fmt.Println(receiver)
	}()

	for range 5 {
		revr := <-greeting
		fmt.Println(revr)
	}

	time.Sleep(1 * time.Second)

	// receiver = <-greeting
	// fmt.Println(receiver)
	fmt.Println("End of program")

}
