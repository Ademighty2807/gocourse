// SYNCHRONIZING DATA EXCHANGE
func main() {
	data := make(chan string)

	go func() {
		for i := 0; i < 5; i++ {
			data <- "hello " + fmt.Sprintf("%d", i)
			time.Sleep(100 * time.Millisecond)
		}
		close(data)
	}()
	// close(data) // Channel closed before Goroutine could send a value to the channel

	for value := range data {
		fmt.Println("Received value:", value, ":", time.Now())
	} // Loop over only active channel, creates receiver each time and stops creating receiver (looping) once the channel is closed
}
