func divide(a, b int) (quotient int, reminder int) {
	quotient = a / b
	reminder = a % b
	return
}

func compare(a, b int) (string, error) {
	if a > b {
		return "a is greater than b", nil
	} else if b > a {
		return " b is greater than a", nil
	} else {
		return "", errors.New("Unable to compare which is greater")
	}
}
