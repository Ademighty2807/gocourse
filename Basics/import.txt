package main

import (
	"fmt"
	ade "net/http"
)

func main() {

	fmt.Println("Hello, Go Standard Library!")

	resp, err := ade.Get("https://jsonplaceholder.typicode.com/posts/1")
	if err != nil {
		fmt.Println("Error:", err)
		return
	}
	defer resp.Body.Close()

	fmt.Println("Response Status:", resp.Status)
}
