func add(x, y int) int {
	return x + y
}

// function that takes another function as a parameter or augment is called higher order function

func applyOperation(x, y int, operation func(int, int) int) int {
	return operation(x, y)
}

// function that returns another function is called closure

func createMultiplier(factor int) func(int) int {
	return func(x int) int {
		return x * factor
	}
}
