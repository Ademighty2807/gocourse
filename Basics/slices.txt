package main

import (
	"fmt"
	"slices"
	// "slices"
)

func main() {
	// var sliceName []ElementType

	// var numbers []int
	// var numbers := []int{1, 2, 3}

	// var numbers := []int{9, 8, 7}

	// slice := make([]int, 5)

	a := [5]int{1, 2, 3, 4, 5}

	slice1 := a[1:4]    // slice1 will contain elements from index 2 to 3 (4 is excluded)
	fmt.Println(slice1) // Output: [3 4]

	slice1 = append(slice1, 6, 7)  // Append 6 and 7 to slice1
	fmt.Println("slice1:", slice1) // Output: [3 4 6 7]

	sliceCopy := make([]int, len(slice1))
	copy(sliceCopy, slice1)
	fmt.Println("sliceCopy:", sliceCopy) // Output: [3 4 6 7]

	// var nilSlice []int
	// if nilSlice == nil {
	// 	fmt.Println("nilSlice is nil") // Output: nilSlice is nil
	// }

	for i, v := range slice1 {
		fmt.Printf("Index: %d, Value: %d\n", i, v)
	}

	fmt.Println("Element of Index 3 of slice1", slice1[3])

	// slice1[3] = 50

	// fmt.Println("Element of Index 3 of slice1:", slice1[3]) // Output: [3 4 6 50]

	if slices.Equal(slice1, sliceCopy) {
		fmt.Println("slice1 and sliceCopy are equal")
	}

	twoD := make([][]int, 3)
	for i := 0; i < 3; i++ {
		innerLen := i + 1
		twoD[i] = make([]int, innerLen)
		for j := 0; j < innerLen; j++ {
			twoD[i][j] = i + j

			fmt.Printf("Adding value %d in outer index %d and inner slice index of %d\n", i+j, i, j)
		}
	}
	fmt.Println("2D slice:", twoD)

	// slice[low:high] // low is inclusive, high is exclusive
	slice2 := slice1[2:3] // slice2 will contain elements from index 1 to 2 (3 is excluded)
	fmt.Println("slice2:", slice2)

	fmt.Println("The capacity of slice2 is", cap(slice2))

	fmt.Println("The length of slice2 is", len(slice2))
}
