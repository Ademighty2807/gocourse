package main

import (
	"embed"
	"fmt"
	"io/fs"
	"log"
)

//go:embed output.txt

var content string

//go:embed basics
var basicFolder embed.FS

func main() {
	fmt.Println("Embedded content:", content)

	content, err := basicFolder.ReadFile("basics/hello.txt")
	if err != nil {
		fmt.Println("Error reading file:", err)
		return
	}
	fmt.Println("Embedded File content:", string(content))

	err = fs.WalkDir(basicFolder, "basics", func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			fmt.Println(err)
			return err
		}
		fmt.Println(path)
		return nil
	})

	if err != nil {
		log.Fatal()
	}
}
